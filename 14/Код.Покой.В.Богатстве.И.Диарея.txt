main.py


import ctypes
import os

# 1. Явная загрузка библиотеки в адресное пространство процесса (аналог dlopen)
lib_path = os.path.abspath("./libmylib.so")
print(f"[ПРОГРАММА] Загружаем внешнюю ассемблерную библиотеку: {lib_path}")

try:
    # Загружаем .so файл. ОС парсит таблицу ELF и проецирует страницы памяти
    my_asm_lib = ctypes.CDLL(lib_path)
except OSError as e:
    print(f"[КРИТИЧЕСКАЯ ОШИБКА]: Не удалось загрузить .so. Проверьте пути. Данные: {e}")
    exit(1)

print("[УСПЕХ] Библиотека спроецирована в память процесса.")

# Подготавливаем тестовые данные (строку)
secret_data = "System Programming 2026"
# Переводим строку в массив сырых байт, так как ассемблер работает с байтами ОЗУ напрямую
binary_buffer = bytearray(secret_data.encode('utf-8'))
buffer_len = len(binary_buffer)

# Получаем низкоуровневый указатель на наш буфер в памяти
c_buffer = (ctypes.c_char * buffer_len).from_buffer(binary_buffer)

print(f"\nИсходная строка в памяти: '{secret_data}'")
print(f"Сырые байты до обработки:  {list(binary_buffer)}")

# 2. Вызываем функцию из библиотеки (аналог dlsym + CALL)
print("\n[ПРОЦЕССОР] Передаем управление в libmylib.so -> функция asm_encrypt...")
my_asm_lib.asm_encrypt(c_buffer, buffer_len)

print(f"Сырые байты после шифрования (XOR): {list(binary_buffer)}")
print(f"Зашифрованная строка в консоли:     '{binary_buffer.decode('utf-8', errors='ignore')}'")

# 3. Повторный вызов той же функции (XOR восстанавливает данные обратно)
print("\n[ПРОЦЕССОР] Повторный вызов asm_encrypt для расшифровки...")
my_asm_lib.asm_encrypt(c_buffer, buffer_len)

print(f"Байты после повторного XOR:         {list(binary_buffer)}")
print(f"Восстановленная строка:            '{binary_buffer.decode('utf-8')}'")

mylib.asm


section .data
    ; Секрет для шифрования. 
    ; Объявляем глобальный символ, чтобы его можно было использовать
    encryption_key db 0x5A    

section .text
    global asm_encrypt

; Функция: asm_encrypt(char* data, int length)
asm_encrypt:
    ; Проверяем, не равна ли длина строки нулю
    cmp rsi, 0
    jle .done
    
    xor rcx, rcx                ; Очищаем регистр-счетчик RCX (индекс символа)

.loop:
    ; --- НИЗКОУРОВНЕВАЯ PIC-АДРЕСАЦИЯ ---
    ; Мы не можем написать напрямую: mov al, [encryption_key]
    ; Мы обязаны использовать адресацию относительно указателя команд RIP!
    mov al, [rel encryption_key] ; Процессор рассчитывает адрес на лету относительно RIP
    
    ; Берем текущий символ строки (адрес строки в RDI + индекс в RCX)
    xor byte [rdi + rcx], al    ; Применяем операцию XOR с ключом
    
    inc rcx                     ; Переходим к следующему символу
    cmp rcx, rsi                ; Сравниваем текущий индекс с длиной строки
    jl .loop                    ; Если индекс меньше длины — крутим цикл дальше

.done:
    ret                         ; Возврат управления в вызывающую программу
